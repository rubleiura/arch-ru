# ℹ️ Назначение: Пошаговая установка Arch Linux с BTRFS. Загрузчик — GRUB,
#    снимки btrfs попадают в меню через grub-btrfs.
# 💡 Метод: Копируйте и вставляйте блоки команд по одному.
# ❗ Важно: Не запускайте как скрипт! Выполняйте вручную.
# 📌 Схема диска: sda1 — BIOS boot (ef02), sda2 — /boot (ext4), sda3 — LUKS2 → LVM vg_main (lv_swap, lv_root → Btrfs)
# 📏 ПРО ОБЪЁМ ВСТАВКИ: тяжёлые блоки помечены окном «ОБЪЁМ БЛОКА N: X КБ
#    ВСТАВКИ». Пока выполняется «тяжёлая» команда (pacman, mkinitcpio,
#    mkfs.btrfs), занятая ВМ может не принять весь вставленный текст —
#    хвост блока теряется. Как не потерять хвост:
#      • работать через SSH в ВМ — надёжнее всего;
#      • вставлять большой блок ЧАСТЯМИ — от одной границы «# #-----» до
#        следующей (границы шагов для этого и нарисованы);
#      • следующую порцию вставлять только ПОСЛЕ завершения тяжёлой команды.
# 🔐 Блоки с интерактивным финалом (cryptsetup…, passwd) устроены так, что
#    после интерактивной команды в блоке НЕ ОСТАЁТСЯ ни одной исполняемой
#    строки: всё вставленное следом ушло бы в ответ на запрос пароля.
# 🌐 Требуется: Интернет, загрузочная среда Arch Linux (свежий ISO).
# 💡 Примечание: Данная установка предназначена для компьютеров с BIOS (Legacy) и полным шифрованием системного раздела.
#-----------------------------------------------------------------
# 🔧 РЕДАКЦИЯ grub-v1.0 (2026-10-08) — собрана движком tools/port_grub.py
#    по эталонам семейства: «BTRFS  UKI  SecureBoot UEFI.txt» (v4.5) и
#    «BTRFS  LUKS2  UKI  SecureBoot UEFI.txt» (luks-v1.3).
# ⚠️ Это ДРУГАЯ ветка семейства. Эталоны работают на systemd-boot + UKI +
#    sbctl (Secure Boot); этот макет — на GRUB + grub-btrfs, и Secure Boot
#    он НЕ поддерживает: GRUB из Arch не подписан, а его модули грузятся
#    с диска непроверяемыми. Нужен Secure Boot — берите UKI-эталон.
# ❗ Поэтому Secure Boot должен быть ВЫКЛЮЧЕН — и в прошивке компьютера,
#    и в настройках ВМ: неподписанный GRUB прошивка не запустит, а ISO
#    Arch может быть отклонён ещё при загрузке (Access denied).
# 🎨 Оформление приведено к эталонам: плоские проверки echo " 🔍 …"
#    заменены цветными окнами с явным «Ожидается: …» и нумерацией
#    «ПРОВЕРКА k/n»; старые баннеры «## ✅ … ЗАВЕРШЕНА ##» — зелёными
#    рамками-итогами «что сделано / куда дальше»; перед каждой
#    интерактивной командой — жёлтое окно-действие. Рамки ровно 76
#    видимых колонок, внутри рамок нет эмодзи, dialog не используется:
#    dialog читает клавиши из stdin и «съедает» хвост вставленного блока.
# 🩹 Исправлены содержательные дефекты исходника (каждый ломал установку
#    или оставлял неработающую функцию):
#    • «🌌 ВАРИАНТ: COSMIC (System76)» без «#» — bash: syntax error near
#      unexpected token '(' — вставка БЛОКА 16 обрывалась;
#    • в HOOKS=(…) возвращён base ПЕРВЫМ хуком: его справка говорит
#      «DO NOT remove this hook», и shipped-дефолт mkinitcpio его содержит;
#    • rm -r ~/.ssh перенесён ДО swapoff/umount/poweroff: стоял после
#      poweroff и не выполнялся никогда;
#    • pacman -Syy → -Syu (3 места): частичное обновление Arch не
#      поддерживает («Partial upgrades are unsupported»);
#    • passwd перенесён в КОНЕЦ БЛОКОВ 10 и 11: после него шло ещё
#      15–16 команд, и весь хвост уходил в ответ на запрос пароля;
#    • timedatectl в chroot заменён на readlink -f /etc/localtime,
#      hwclock --show и date: в chroot нет активного D-Bus, и команда
#      падала — пользователь думал, что установка сломана;
#    • из mkdir убраны var/lib/machines и var/lib/portables: они ничем
#      не монтировались и попадали внутрь @ и в каждый снимок snapper;
#    • mkfs.btrfs -f → -f -K: без -K команда делает TRIM всего раздела
#      и идёт долго — на занятой ВМ хвост вставки терялся;
#    • параметры своппинга пересобраны: vm.swappiness=10 из исходника
#      заменён на 100 и дополнен vm.watermark_boost_factor=0,
#      vm.watermark_scale_factor=125 и vm.page-cluster=0 (ArchWiki → Zram).
#      Причина: swap в этой ветке двойной — реальный диск для гибернации
#      и zram в БЛОКЕ 13 ШАГ 8 с приоритетом 100. При swappiness=10 ядро
#      почти не использует сжатую память и смещает давление в earlyoom —
#      та же ошибка, которую эталон v4.5 исправил значением 180;
#    • раскладка консоли приведена к us/us: loadkeys us в БЛОКЕ 2 и
#      KEYMAP=us в БЛОКЕ 9 (в незашифрованном эталоне v4.5 — KEYMAP=ru;
#      решение принято для единообразия семейства и безопасно: здесь
#      ранняя загрузка пароль не спрашивает);
#    • добавлен nftables: fail2ban не является фильтром пакетов, а при
#      включённых sshd, avahi и cups система была открыта в сеть;
#    • включён fstrim.timer — страховка на случай, если токен «defaults»
#      в БЛОКЕ 3 оставлен как есть (иначе TRIM не выполнялся бы вовсе);
#    • «quiet splash» перенесён в БЛОК 18: при неудачной первой загрузке
#      причину нельзя прятать за заставкой plymouth;
#    • пересажен из эталонов шаг «PLYMOUTH — ТЕМА ЗАСТАВКИ»:
#      /etc/plymouth/plymouthd.conf (Theme=bgrt, ShowDelay=0) — без него
#      заставка ждёт 5 секунд и на быстрой загрузке не появляется вовсе
#      (на ВМ это выглядело как «plymouth не работает»);
#    • из HOOKS незашифрованных редакций убран keyboard (канон эталона
#      v4.5: запроса пароля в ранней загрузке нет), из комментариев шага —
#      чужой контекст «загрузка с зашифрованного LVM тома»;
#    • sed-и GRUB_BTRFS_SUBMENUNAME/TITLE_FORMAT приведены к рабочей форме:
#      в незашифрованных редакциях кавычки не были экранированы — bash рвал
#      команду на слова, sed падал («unterminated `s' command»), и подменю
#      «Arch Linux snapshots» в меню не появлялось;
#    • из БЛОКА 9 убрана «правка fstab» subvol=…,defaults: незашифрованная
#      форма выполнялась, но затирала ВСЕ выбранные опции монтирования, а
#      LUKS-форма падала («invalid reference \1»). Как и в эталонах, fstab
#      остаётся таким, каким его записал genfstab;
#    • семь вызовов pacstrap БЛОКА 8 собраны в ОДИН «pacstrap -K /mnt …» —
#      форма эталонов: каждый отдельный вызов заново синхронизировал базы
#      пакетов (pacman -Sy) и заново разбирался с keyring; -K создаёт keyring
#      заново в целевой системе, а не копирует его из live-ISO;
#    • LUKS BIOS: plymouth ставился дважды (общей строкой и отдельно),
#      wpa_supplicant/wireless_tools отсутствовали;
#    • нумерация шагов БЛОКА 12 приведена к единой для всех редакций:
#      в Arch UEFI было два «ШАГ 5», в LUKS BIOS раздел GRUB-BTRFS стоял
#      без заголовка;
#    • разделители шагов и линейки блоков приведены к каноническому виду
#      семейства: «# #-----…» (шаг) и «####…» (блок).
# 🆕 Добавлен БЛОК 18 «ПОСЛЕ ПЕРВОЙ УСПЕШНОЙ ЗАГРУЗКИ»: диагностика
#    systemd, включение тихой загрузки, первое pacman -Syu, проверка
#    снимков snapper, подтомов, таймеров, nftables и swap; после включения
#    тихой загрузки проверяется, что «quiet splash» попал в grub.cfg, а
#    ПОСЛЕ перезагрузки — что параметры дошли до ядра (/proc/cmdline) и что
#    plymouth действительно запускался (journalctl -b): правильные файлы
#    загрузку не доказывают, там же — разбор причин и откат к подробному логу.
# ℹ️ ЧТО НЕ МЕНЯЛОСЬ НАМЕРЕННО (специфика GRUB-ветки, переносу из
#    эталонов не подлежит): реальный swap на диске и гибернация через
#    resume=UUID=…, grub-btrfs/grub-btrfsd, memtest через grub-mkconfig,
#    GRUB_ENABLE_CRYPTODISK=y как страховка.
#
# Ключевые решения:
# • Загрузчик — GRUB + grub-btrfs (НЕ systemd-boot + UKI). Secure Boot эта
#   ветка НЕ поддерживает: GRUB из Arch не подписан, а его модули грузятся
#   с диска непроверяемыми. Нужен Secure Boot — берите UKI-эталон семейства.
# • Ядро — linux-lts. Параметры ядра прописываются в /etc/default/grub и
#   попадают в меню через grub-mkconfig.
# • Корень монтируется ЯВНО подтомом @ (subvol=@ в fstab и в параметрах
#   ядра). Подтом по умолчанию НЕ меняется: иначе «Восстановить копию» в
#   btrfs-assistant (переименование подтомов) оставит систему загруженной
#   из backup-копии.
# • Снимки btrfs в меню — grub-btrfs + grub-btrfsd.service: пункты появляются
#   автоматически при создании снимка.
# • Swap РЕАЛЬНЫЙ, на диске: resume=UUID=… даёт гибернацию (в UKI-эталонах
#   swap целиком в zram). При этом zram в БЛОКЕ 13 ШАГ 8 тоже включён и у него
#   swap-priority=100 — то есть swap двойной.
# • Параметры своппинга: vm.swappiness=100 (среднее между 10 из исходника и
#   180 из эталона v4.5), vm.watermark_boost_factor=0,
#   vm.watermark_scale_factor=125, vm.page-cluster=0 — по ArchWiki (Zram).
#   При swappiness=10 ядро почти не использовало бы сжатую память и смещало
#   всё давление в earlyoom; page-cluster=0 убирает чтение страниц пачками по 8.
# • quiet splash loglevel=3 — ТОЛЬКО в БЛОКЕ 18, после первой успешной
#   загрузки: иначе причину сбоя не увидеть за заставкой plymouth.
# • Plymouth: /etc/plymouth/plymouthd.conf с Theme=bgrt и ShowDelay=0 —
#   шаг пересажен из эталонов. Без ShowDelay=0 заставка ждёт 5 секунд и на
#   быстрой загрузке (ВМ) не появляется вовсе. Файл создаётся ДО mkinitcpio -P:
#   хук plymouth упаковывает его внутрь initramfs. В VirtualBox нет таблицы
#   BGRT — будет только анимация, это норма (как в эталонах).
# • Проверка ПОСЛЕ перезагрузки — последний шаг БЛОКА 18. «quiet splash» в
#   /etc/default/grub и в grub.cfg — это только обещание: загрузчик мог
#   выбрать другую запись, параметры могли не дойти до ядра, а plymouth —
#   не стартовать при живом хуке. Достоверный ответ даёт работающая система:
#   /proc/cmdline (что ядру реально передано) и journalctl -b (что реально
#   делал plymouth). Команда reboot исполняемой строкой в блок НЕ вставлена:
#   вставка оборвалась бы на середине, и шаг не выполнился бы никогда.
# • nftables вместо «защиты сети» через fail2ban: fail2ban банит по логам
#   и НЕ является фильтром пакетов, а sshd, avahi и cups были открыты.
# • fstrim.timer включён как страховка: если токен «defaults» в БЛОКЕ 3
#   оставлен как есть, discard=async не включится и TRIM не будет вовсе.
# • BIOS-загрузка: раздел ef02 (BIOS boot, 4 МБ) не форматируется и не
#   монтируется — в него grub-install --target=i386-pc кладёт core.img.
# • memtest86+ (без -efi): пункт меню создаёт grub-mkconfig.
# • Шифрование: один контейнер LUKS2 с меткой cryptlvm, внутри — LVM
#   (vg_main: lv_swap 4G, lv_root), внутри lv_root — Btrfs с подтомами.
#   /boot НЕ шифруется: GRUB должен прочитать ядро до ввода пароля.
# • Копия заголовка LUKS делается дважды — в БЛОКЕ 7B и в БЛОКЕ 18: после
#   luksAddKey заголовок другой, и первая копия уже не актуальна. Храните
#   копию ВНЕ этого диска: ключи лежат в заголовке, его повреждение =
#   потеря данных при правильном пароле.
# • Второй слот ключа (БЛОК 7B) обязателен: забытый единственный пароль =
#   потеря данных.
# • Хуки initramfs: base systemd … sd-encrypt lvm2 resume filesystems fsck.
#   Порядок критичен: sd-encrypt ДО lvm2 (сначала открывается контейнер),
#   resume ПОСЛЕ lvm2 (swap-том активируется до восстановления).
# • Раскладка консоли — us (решение эталона luks-v1.3, подтверждено
#   владельцем): loadkeys us в live-среде и KEYMAP=us в /etc/vconsole.conf.
#   Этот файл читает sd-vconsole в ранней загрузке, поэтому пароль LUKS
#   создаётся и вводится латиницей — и его можно ввести с флешки
#   восстановления, где раскладка us по умолчанию. Русский вывод в консоли
#   сохраняет шрифт cyr-sun16; для русского ввода в live выполните loadkeys ru
#   ДО БЛОКА 7A — но тогда и пароль придётся задавать кириллицей.

# 💡 Принцип: Скрипт определяет ближайшие страны → вы заменяете

# 🧹 Очистка экрана терминала
clear
# 🎨 Цвета окон и рамок (безопасно задавать повторно в каждом блоке)
CN=$'\e[0m'; CB=$'\e[1;36m'; CW=$'\e[1;37m'; CG=$'\e[1;32m'; CY=$'\e[1;33m'; CR=$'\e[1;31m'; TB=$'\e[1;30;38;2;0;0;0;46m'; TG=$'\e[1;30;38;2;0;0;0;42m'; TY=$'\e[1;30;38;2;0;0;0;43m'
# 🏔️ Приветственный баннер: логотип Arch Linux
#------------------------------------------------------------------------------
# 💡 Печатается сразу после clear — начало работы с макетом видно издалека.
#    Цвет — «фирменный» cyan Arch (как рисует neofetch): задаётся одним printf
#    до вывода арта и сбрасывается после, поэтому раскрашивать построчно не
#    нужно. Heredoc << 'EOF' закавычен: обратные кавычки внутри логотипа —
#    обычные символы, команды не выполняются. Ничего не читает stdin — вставку
#    блока не ломает.
LA=$'\e[1;36m'   # Arch-cyan — логотип
LW=$'\e[1;37m'   # яркий белый — заголовок
# 🎨 CN здесь намеренно не задаётся: строка определений цветов в
#    макете начинается с «CN=$'», и по этому признаку движок
#    понимает, что цвета в блоке уже заданы. Полный набор
#    (CB/CG/CY/TB/TG/TY) ставится перед первым окном блока.
printf '%s' "$LA"
cat << 'EOF'
                                        -`
                                       .o+`
                                      `ooo/
                                     `+oooo:
                                    `+oooooo:
                                    -+oooooo+:
                                  `/:-:++oooo+:
                                 `/++++/+++++++:
                                `/++++++++++++++:
                               `/+++ooooooooooooo/`
                              ./ooosssso++osssssso+`
                             .oossssso-````/ossssss+`
                            -osssssso.      :ssssssso.
                           :osssssss/        osssso+++.
                          /ossssssss/        +ssssooo/-
                        `/ossssso+/:-        -:/+osssso+-
                       `+sso+:-`                 `.-/+oso:
                      `++:.                           `-/+/
                      .`                                 `/
EOF
printf '%s\n' "${LW}              A R C H   L I N U X   ·   Б Л О Ч Н Ы Й   М А К Е Т${CN}"

def haversine(lat1, lon1, lat2, lon2):
    R = 6371.0
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    a = math.sin(dlat/2)**2 + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlon/2)**2
    return R * 2 * math.atan2(math.sqrt(a), math.sqrt(1-a))

def get_ip_location():
    apis = [
        ("https://ipapi.co/json/", "ipapi.co"),
        ("https://ipinfo.io/json", "ipinfo.io"),
    ]
    for api_url, api_name in apis:
        try:
            print(f"  🔄 Пробуем {api_name}...")
            with urllib.request.urlopen(api_url, timeout=5) as resp:
                data = json.loads(resp.read().decode())

                # ✅ ИСПРАВЛЕНО: Добавлена переменная data в условия
                if api_name == "ipapi.co":
                    if "error" in data:
                        continue
                    if "latitude" not in data or "longitude" not in data:
                        continue
                    lat = float(data.get("latitude", 0))
                    lon = float(data.get("longitude", 0))
                    country = data.get("country_name", "Unknown")

                elif api_name == "ipinfo.io":
                    if "bogon" in data:
                        continue
                    if "loc" not in data:
                        continue
                    loc = data.get("loc", "0,0").split(",")
                    lat = float(loc[0])
                    lon = float(loc[1])
                    country = data.get("country", "Unknown")
                else:
                    continue

                if lat != 0 and lon != 0:
                    print(f"  ✅ Использован API: {api_name}")
                    return lat, lon, country

        except Exception as e:
            print(f"  ⚠️ {api_name} недоступен: {e}")
            continue

    print("❌ Все API недоступны. Проверьте подключение к интернету.")
    sys.exit(1)

def main():
    print("🌐 Определение местоположения по IP...")
    my_lat, my_lon, my_country = get_ip_location()
    print(f"✅ Вы находитесь: {my_country} ({my_lat:.4f}, {my_lon:.4f})\n")

    distances = []
    for country, (cap_lat, cap_lon) in CAPITALS.items():
        if country == my_country:
            continue
        dist = haversine(my_lat, my_lon, cap_lat, cap_lon)
        distances.append((country, dist))

    distances.sort(key=lambda x: x[1])

    print(f"{"#":<3} {"Страна":<25} {"Расстояние (км)":<10}")
    print("-" * 45)
    for i, (country, dist) in enumerate(distances[:10], 1):
        print(f"{i:<3} {country:<25} {dist:.0f}")
    print("-" * 45)
    print("✅ Готово! Используйте список выше для настройки.")

if __name__ == "__main__":

#       cryptsetup luksFormat --type luks2 --label cryptlvm --pbkdf-memory 262144 /dev/sda3

# 🔐 /etc/crypttab НЕ нужен и не создаётся: корневой контейнер cryptlvm
#    открывает initramfs по параметру ядра rd.luks.name=<UUID>=cryptlvm

#    после открытия контейнера этот UUID всегда доступен системе.
# 📋 Закрываем ESP от остальных пользователей. FAT не хранит права Unix —
#    их задают маски монтирования. Без этого /boot доступен всем локальным
#    пользователям, то есть любой из них может подменить ядро или initramfs
#    и получить выполнение своего кода при следующей загрузке.
# ℹ️ В эталоне v4.5 обоснование иное: там systemd-boot держит на ESP

# ⚠️ Побочный эффект: после этого обычный пользователь НЕ СМОЖЕТ читать /boot —

# #------------------------------------------------------------------------------
# ⌨️ Раскладка консоли — us (латинская), шрифт — cyr-sun16 (с кириллицей).
#    ❗ Это ОСОЗНАННОЕ решение, то же, что в эталоне luks-v1.3 (там KEYMAP=us
#    вместо KEYMAP=ru незашифрованной редакции). Причина: пароль LUKS вводится
#    в ранней загрузке, а раскладка туда берётся из этого же файла. При
#    KEYMAP=ru запрос пароля ждал бы кириллицу, а с флешки восстановления
#    (там раскладка us по умолчанию) ввести её было бы нельзя —
#    «No key available with this passphrase». Одна раскладка везде = нет
#    путаницы: пароль всегда вводится латиницей.
#    💡 Русский ВЫВОД в консоли сохраняется — за кириллицу отвечает ШРИФТ, а не
#    раскладка. Нужен русский ВВОД в консоли — выполните loadkeys ru (но тогда
#    и пароль LUKS придётся вводить кириллицей, и KEYMAP менять уже нельзя).
#    В графической среде раскладка настраивается отдельно и на LUKS не влияет.
#    💡 АЛЬТЕРНАТИВА, если русский ВВОД в консоли всё же нужен: оставьте здесь
#       KEYMAP=ru, а в GRUB_CMDLINE_LINUX_DEFAULT (БЛОК 12) добавьте
#       vconsole.keymap=us. Параметр задаст раскладку только для ранней
#       загрузки (запрос пароля), а установленная система останется русской.
#       Макет так не делает СОЗНАТЕЛЬНО: две разные раскладки — ровно та
#       путаница, из-за которой пароль вводят не тем алфавитом.
#    💡 Русскую раскладку для графики можно задать одной командой (её читают
#       X11/XWayland и большинство DE; в чистом Wayland надёжнее настроить в
#       параметрах самой среды — БЛОК 16):
#          localectl set-x11-keymap ru,us grp:alt_shift_toggle
#       Команда намеренно НЕ выполняется в макете: среда всё равно переспросит
#       раскладку при первой настройке, а навязывать её мы не хотим.

# ШАГ 3: НАСТРОЙКА MKINITCPIO (HOOKS И MODULES)
# #------------------------------------------------------------------------------
# ℹ️ Используются systemd-хуки (нужны для plymouth, шифрования и снимков):
#    base — остаётся первым, как в ArchWiki;  systemd — заменяет udev;
#    microcode — ОБЯЗАТЕЛЕН: микрокод упаковывается внутрь initramfs;
#    keyboard — драйверы клавиатуры в ранней загрузке: без них пароль LUKS
#      будет некому ввести. СОЗНАТЕЛЬНО поставлен ДО autodetect — так советует
#      ArchWiki (dm-crypt/System configuration): «it is helpful to place this
#      hook before autodetect in order to always include all keyboard drivers.
#      Otherwise the external keyboard only works in early userspace if it was
#      connected when creating the image». Иначе в образ попадают драйверы
#      только той клавиатуры, что была подключена в момент сборки. Сценарий
#      поломки: установка в VirtualBox или с внутренней клавиатурой → загрузка
#      с внешней USB-клавиатурой → пароль ввести НЕЧЕМ;
#    sd-vconsole — раскладка и шрифт консоли, в том числе для запроса пароля;
#      ОБЯЗАТЕЛЬНО ДО sd-encrypt: к запросу пароля применяется KEYMAP из
#      /etc/vconsole.conf (у нас us — см. БЛОК 9);
#    plymouth — ДО block и sd-encrypt, чтобы заставка появилась раньше запроса
#      пароля (иначе пароль пришлось бы вводить на чёрном экране);
#    sd-encrypt — systemd-вариант хука encrypt: открывает cryptlvm по
#      rd.luks.name=… из параметров ядра. Именно sd-encrypt, НЕ encrypt:
#      только он работает с plymouth и с systemd-генераторами;
#    lvm2 — ПОСЛЕ sd-encrypt и ДО filesystems: сначала открывается контейнер,
#      потом активируются тома;
#    resume — ОБЯЗАТЕЛЕН, раз в параметрах ядра есть resume=UUID=…, и стоит
#      ПОСЛЕ lvm2: swap-том lv_swap должен быть активирован до восстановления.
#      Без resume гибернация молча не работает — initramfs даже не пытается
#      восстановиться из swap.
sed -i "s|^HOOKS=(.*)|HOOKS=(base systemd keyboard autodetect microcode modconf kms sd-vconsole plymouth block sd-encrypt lvm2 resume filesystems fsck)|" /etc/mkinitcpio.conf
# 📋 Явно добавляем модули btrfs и dm-crypt — надёжнее, чем полагаться на
#    autodetect (в VirtualBox autodetect иногда не добирает dm_*).
sed -i "s/MODULES=()/MODULES=(btrfs dm_mod dm_crypt)/" /etc/mkinitcpio.conf

#    таблицу BGRT). В VirtualBox BGRT нет, поэтому будет только анимация — это норма.
# ❗ Файл создаётся ДО пересборки initramfs («ШАГ 7»): хук plymouth
#    упаковывает plymouthd.conf внутрь образа. Без этого файла тема остаётся
#    заводской (spinner), а заставка ждёт ShowDelay=5 с — на быстрой загрузке
#    она не появляется вовсе: именно так на ВМ выглядело «plymouth не работает».

# ШАГ 5: НАСТРОЙКА ПАРАМЕТРОВ ЯДРА (GRUB)
# #------------------------------------------------------------------------------
# ℹ️ Получение UUID разделов для параметров ядра.
# ❗ Убедитесь, что имена разделов (sda2, sda3) соответствуют вашему диску!
# #------------------------------------------------------------------------------
# 📋 Получение UUID LUKS контейнера
CRYPT_UUID=$(blkid -s UUID -o value /dev/sda3)
# 📋 Получение UUID swap раздела
SWAP_UUID=$(blkid -s UUID -o value /dev/vg_main/lv_swap)
# 📋 Получение UUID root раздела
ROOT_UUID=$(blkid -s UUID -o value /dev/vg_main/lv_root)
# 📋 Настройка GRUB_CMDLINE_LINUX_DEFAULT
# ✅ Замена всей строки с параметрами загрузки.
# ❗ БЕЗ «quiet splash»: при неудачной первой загрузке причину нельзя
#    прятать за заставкой plymouth. Тихая загрузка включается в БЛОКЕ 18 —
#    после того, как система успешно загрузилась хотя бы раз.
sed -i "s|^GRUB_CMDLINE_LINUX_DEFAULT=.*|GRUB_CMDLINE_LINUX_DEFAULT=\"loglevel=3 resume=UUID=${SWAP_UUID}\"|" /etc/default/grub
# 📋 Настройка GRUB_CMDLINE_LINUX
# ❗ rd.lvm.lv=… НЕ добавляется намеренно: с systemd-инитрамфсом хук lvm2
#    активирует группы томов сам, а ArchWiki (dm-crypt/Encrypting an entire
#    system, раздел «LVM on LUKS») приводит для загрузчика ровно
#    rd.luks.name=<UUID>=cryptlvm и root=UUID=<lv_root>.
#    rd.lvm.lv=… — синтаксис dracut, в Arch он не нужен.
# 📋 rd.luks.options=discard — каноническая часть дельты шифрования
#    эталона luks-v1.3. Без него TRIM молча теряется на dm-crypt:
#    fstrim.timer из БЛОКА 18 чистит блоки внутри btrfs, но запрос не
#    доходит до SSD, и ресурс накопителя расходуется впустую.
#    Честно о расплате: discard раскрывает сведения о занятом месте
#    на контейнере — эталон принимает этот компромисс осознанно.
sed -i "s|^GRUB_CMDLINE_LINUX=.*|GRUB_CMDLINE_LINUX=\"rd.luks.name=${CRYPT_UUID}=cryptlvm rd.luks.options=discard root=UUID=${ROOT_UUID}\"|" /etc/default/grub
# 📋 Включение поддержки шифрованного диска в GRUB
# ℹ️ GRUB_ENABLE_CRYPTODISK=y оставлен как СТРАХОВКА: /boot в этой схеме
#    НЕ зашифрован (ESP или отдельный ext4-раздел), поэтому GRUB не обязан
#    уметь читать LUKS. Параметр действительно нужен, если /boot лежит
#    внутри контейнера — тогда GRUB сам спросит пароль до загрузки ядра.
sed -i "s|^# GRUB_ENABLE_CRYPTODISK=.*|GRUB_ENABLE_CRYPTODISK=y|" /etc/default/grub

# 📋 Параметры своппинга (ArchWiki → Zram, § «Optimizing swap on zram»).
# ❗ В исходнике здесь стояло vm.swappiness=10 с обоснованием «рекомендуется
#    для SSD». Обоснование неполное: swap в этой ветке ДВОЙНОЙ — реальный
#    дисковый (гибернация через resume=UUID=…) и zram в ШАГЕ 8 ниже, причём
#    у zram swap-priority=100, то есть он основной. При swappiness=10 ядро
#    почти не использует сжатую память и смещает всё давление в
#    earlyoom/OOM-killer — ровно та ошибка, которую эталон v4.5 исправил,
#    подняв значение до 180 (в UKI-ветке дискового swap нет вовсе).
#    Здесь взято 100 — среднее: zram работает, а своппинг на диск
#    остаётся умеренным (решение владельца, вопрос 4 журнала).
cat > /etc/sysctl.d/99-swap-parameters.conf << "EOF"
vm.swappiness = 100
# kswapd начинается раньше и работает дольше — меньше риск прямого
# reclaim (подвисаний) при переполнении памяти:
vm.watermark_boost_factor = 0
vm.watermark_scale_factor = 125
# читать по одной странице, а не пачками по 8: и для zram, и для SSD
# пачка — лишние задержки и лишний трафик:
vm.page-cluster = 0
EOF

pacman -S --noconfirm nftables
# 📋 Декларативный конфиг: входящие запрещены по умолчанию; разрешены
#    установленные/родственные соединения, loopback, ICMP и SSH с
#    ограничением 4 новых подключения в минуту с адреса.
cat > /etc/nftables.conf << "EOF"
table inet arch_install {
    chain input {
        type filter hook input priority 0; policy drop;

        # Established/related — ответы на наши же исходящие соединения
        ct state established,related accept;
        ct state invalid drop;

        # Loopback
        iif "lo" accept;

        # ICMP/ICMPv6 — без него сеть «молча не пингуется»
        ip protocol icmp accept;
        ipv6-nexthdr icmpv6 accept;

        # SSH: не более 4 новых подключений в минуту с одного адреса
        tcp dport 22 ct state new meter ssh_limit { ip saddr limit rate 4/minute burst 4 packets } accept;

        # Локальная сеть (avahi/mDNS, cups): разрешаем только link-local
        udp dport 5353 ip saddr 169.254.0.0/16 accept;

        log prefix "nftables-drop: " limit rate 5/minute;
    }

    chain forward {
        type filter hook forward priority 0; policy drop;
    }

    chain output {
        type filter hook output priority 0; policy accept;
    }
}
EOF

#    загрузке; проверка — БЛОК 18 (sudo nft list ruleset).

# 📌 ГЛАВНАЯ КОМАНДА ШАГА: что загрузчик ДЕЙСТВИТЕЛЬНО передал ядру.
#    До перезагрузки этот вопрос задать нельзя — файла с ответом нет.
cat /proc/cmdline
# 🔍 Вердикт одной строкой. quiet гасит лог ядра, splash разрешает
#    заставку plymouth — нужны ОБА параметра, поэтому проверяются оба.
grep -qw quiet /proc/cmdline && grep -qw splash /proc/cmdline && echo "✅ ТИХАЯ ЗАГРУЗКА РАБОТАЕТ: ядро получило quiet splash" || echo "❗ Ядро запущено БЕЗ quiet splash — лог будет виден. Порядок действий — в комментариях ниже"
# 📋 Вторая половина вопроса — заставка. Параметры в cmdline ещё не
#    означают, что plymouth стартовал: хук мог не попасть в initramfs,
#    а ShowDelay=5 (заводское значение) — не дать заставке появиться на
#    быстрой загрузке. Факт запуска виден в журнале ЭТОЙ загрузки (-b).
sudo journalctl -b --no-pager | grep -i -m5 plymouth || echo "❗ В журнале этой загрузки нет ни одной строки plymouth — заставка не запускалась (порядок действий — в комментариях ниже)"
# 📋 Тема заставки (ожидается bgrt) и её конфиг: ShowDelay=0 обязателен,
#    иначе plymouth ждёт 5 секунд и на быстрой загрузке не появляется
#    вовсе — именно так это выглядело как «plymouth вообще не работает».
plymouth-set-default-theme

#    будет только анимация (см. окно выше). На реальном UEFI-ПК здесь
#    появится путь к логотипу производителя.
